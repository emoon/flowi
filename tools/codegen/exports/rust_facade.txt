# Internal symbols that flowi's own Rust facade wraps. They are declared in internal
# headers under src/, have no .def entry, and are not public C API. They are exported
# because the Rust crates link libflowi.so, and a safe wrapper in the `flowi` crate
# cannot link against a symbol the export list omits.
#
# An entry leaves when its facade does, or when the API moves into the IDL and api_gen
# emits it into flowi.version. Each entry widens libflowi.so's ABI, so add one only for
# a symbol a flowi-side facade needs.
#
# Same format as handwritten.txt: api_gen merges it into the export list,
# one symbol per line, `#` starts a comment, every entry a C identifier.

# core/cpu_topology.h: behind `flowi::performance_cores` / `flowi::pin_current_thread`.
fl_cpu_query_by_type
fl_cpu_set_thread_affinity

# core/hash.h: behind the `flowi::hash` module (`crc32`, `sha1`, `md5`, `Sha256`).
fl_hash_crc32
fl_hash_md5
fl_hash_sha1
fl_hash_sha256_digest_hex
fl_hash_sha256_final
fl_hash_sha256_hex
fl_hash_sha256_init
fl_hash_sha256_update

# vfs/vfs.h: behind `flowi::Vfs::init` and `flowi::Mount::job_handle`. The rest of the
# VFS is public .def API; these two leaves are not.
fl_vfs_init
fl_vfs_mount_get_job_handle
